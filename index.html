  <!-- 카카오 지도 SDK: 장소 검색을 위해 services 라이브러리 포함 -->

  <script

    src="https://dapi.kakao.com/v2/maps/sdk.js?appkey=a37189a9faf51c8c9c2b9678257633c4&libraries=services&autoload=false"

    onerror="showSdkError()"

  ></script>


  <script>

    // 지도와 검색에 필요한 변수를 선언합니다.

    let map;

    let placesService;

    let infoWindow;

    let markers = [];

    let currentPosition = null;

    let mapReady = false;


    const mapElement = document.getElementById("map");

    const searchForm = document.getElementById("searchForm");

    const keywordInput = document.getElementById("keyword");

    const statusElement = document.getElementById("status");

    const resultList = document.getElementById("resultList");


    // 안내 문구를 변경합니다.

    function setStatus(message) {

      statusElement.textContent = message;

    }


    // SDK를 불러오지 못했을 때 안내합니다.

    function showSdkError() {

      setStatus(

        "지도 SDK를 불러오지 못했습니다. 인터넷 연결과 카카오 키 설정을 확인해 주세요."

      );

    }


    // 지도를 생성한 뒤 나머지 기능을 실행합니다.

    function initializeMap() {

      if (!window.kakao || !kakao.maps) {

        showSdkError();

        return;

      }


      try {

        // 처음에는 서울 시청을 기본 위치로 사용합니다.

        const initialPosition = new kakao.maps.LatLng(

          37.5665,

          126.9780

        );


        // 지도 확대 레벨을 5로 설정합니다.

        map = new kakao.maps.Map(mapElement, {

          center: initialPosition,

          level: 5

        });


        // 장소 검색 서비스와 정보 창을 생성합니다.

        placesService = new kakao.maps.services.Places(map);

        infoWindow = new kakao.maps.InfoWindow({

          zIndex: 1

        });


        // 지도가 준비된 다음 위치 확인과 검색을 시작합니다.

        kakao.maps.event.addListener(map, "tilesloaded", function () {

          if (mapReady) return;


          mapReady = true;

          setStatus("현재 위치를 확인하고 있습니다.");

          requestCurrentPosition();

        });


      } catch (error) {

        console.error("지도 초기화 오류:", error);

        setStatus(

          "지도 초기화에 실패했습니다. JavaScript 키와 웹 도메인을 확인해 주세요."

        );

      }

    }


    // 브라우저에 현재 위치 권한을 요청합니다.

    function requestCurrentPosition() {

      if (!navigator.geolocation) {

        setStatus("현재 위치 기능을 지원하지 않는 브라우저입니다.");

        return;

      }


      navigator.geolocation.getCurrentPosition(

        function (position) {

          // 위치 확인에 성공하면 지도 중심을 현재 위치로 이동합니다.

          const latitude = position.coords.latitude;

          const longitude = position.coords.longitude;


          currentPosition = new kakao.maps.LatLng(

            latitude,

            longitude

          );


          map.setCenter(currentPosition);

          map.setLevel(5);


          // 현재 위치를 지도에 표시합니다.

          new kakao.maps.Marker({

            map: map,

            position: currentPosition,

            title: "현재 위치"

          });


          setStatus("현재 위치를 표시했습니다. 장소를 검색해 보세요.");

        },

        function (error) {

          // 위치 권한을 거부하거나 위치를 확인하지 못한 경우입니다.

          console.warn("현재 위치 확인 실패:", error);


          if (error.code === 1) {

            setStatus(

              "위치 권한이 거부되었습니다. 서울 시청을 기준으로 장소를 검색할 수 있습니다."

            );

          } else if (error.code === 2) {

            setStatus(

              "현재 위치를 확인할 수 없습니다. 서울 시청을 기준으로 검색합니다."

            );

          } else {

            setStatus(

              "위치 확인 시간이 초과되었습니다. 서울 시청을 기준으로 검색합니다."

            );

          }

        },

        {

          enableHighAccuracy: false,

          timeout: 10000,

          maximumAge: 60000

        }

      );

    }


    // 이전 검색에서 표시한 마커를 모두 제거합니다.

    function clearMarkers() {

      markers.forEach(function (marker) {

        marker.setMap(null);

      });


      markers = [];

      infoWindow.close();

    }


    // 장소 이름을 HTML에 안전하게 표시하기 위해 문자를 변환합니다.

    function escapeHtml(value) {

      return String(value ?? "").replace(/[&<>"']/g, function (char) {

        const entities = {

          "&": "&amp;",

          "<": "&lt;",

          ">": "&gt;",

          '"': "&quot;",

          "'": "&#39;"

        };


        return entities[char];

      });

    }


    // 입력한 키워드로 장소를 검색합니다.

    function searchPlaces(keyword) {

      if (!mapReady || !placesService) {

        setStatus("지도가 준비될 때까지 잠시 기다려 주세요.");

        return;

      }


      const searchKeyword = keyword.trim();


      if (!searchKeyword) {

        setStatus("검색할 장소를 입력해 주세요.");

        keywordInput.focus();

        return;

      }


      setStatus("장소를 검색하고 있습니다.");

      resultList.replaceChildren();

      clearMarkers();


      // 현재 위치가 확인되었다면 현재 위치 주변을 우선 검색합니다.

      const searchOptions = currentPosition

        ? {

            location: currentPosition,

            sort: kakao.maps.services.SortBy.DISTANCE

          }

        : {};


      placesService.keywordSearch(

        searchKeyword,

        function (data, status) {

          // 검색이 성공한 경우 결과를 표시합니다.

          if (status === kakao.maps.services.Status.OK) {

            displayPlaces(data);

            setStatus("검색 결과 " + data.length + "개");


          } else if (

            status === kakao.maps.services.Status.ZERO_RESULT

          ) {

            setStatus("검색 결과가 없습니다. 다른 키워드를 입력해 보세요.");


          } else {

            setStatus("검색 중 오류가 발생했습니다. 잠시 후 다시 시도해 주세요.");

          }

        },

        searchOptions

      );

    }


    // 검색 결과를 목록과 지도 마커로 표시합니다.

    function displayPlaces(places) {

      const bounds = new kakao.maps.LatLngBounds();


      places.forEach(function (place, index) {

        const position = new kakao.maps.LatLng(

          Number(place.y),

          Number(place.x)

        );


        // 지도에 장소 마커를 추가합니다.

        const marker = new kakao.maps.Marker({

          map: map,

          position: position,

          title: place.place_name

        });


        markers.push(marker);

        bounds.extend(position);


        // 검색 결과 목록 항목을 생성합니다.

        const item = document.createElement("article");

        item.className = "result-item";

        item.tabIndex = 0;

        item.setAttribute("role", "button");

        item.setAttribute(

          "aria-label",

          place.place_name + " 지도에서 보기"

        );


        item.innerHTML = `

          <div class="place-name">

            ${index + 1}. ${escapeHtml(place.place_name)}

          </div>

          <div class="place-category">

            ${escapeHtml(place.category_name || "분류 정보 없음")}

          </div>

          <div class="place-address">

            ${escapeHtml(place.road_address_name || place.address_name)}

          </div>

          <div class="place-phone">

            ${escapeHtml(place.phone || "전화번호 정보 없음")}

          </div>

          <a

            class="place-link"

            href="${escapeHtml(place.place_url || "https://map.kakao.com/")}"

            target="_blank"

            rel="noopener noreferrer"

          >카카오맵에서 보기 ↗</a>

        `;


        // 목록 항목을 클릭하면 해당 장소로 지도를 이동합니다.

        function selectPlace() {

          map.panTo(position);


          infoWindow.setContent(

            '<div style="padding:10px;font-size:13px;">' +

            escapeHtml(place.place_name) +

            "</div>"

          );


          infoWindow.open(map, marker);

        }


        item.addEventListener("click", function (event) {

          // 링크를 클릭했을 때는 지도 이동 이벤트를 실행하지 않습니다.

          if (event.target.closest("a")) return;

          selectPlace();

        });


        item.addEventListener("keydown", function (event) {

          if (event.key === "Enter" || event.key === " ") {

            event.preventDefault();

            selectPlace();

          }

        });


        // 지도 마커를 클릭해도 장소 이름이 표시됩니다.

        kakao.maps.event.addListener(marker, "click", function () {

          selectPlace();

        });


        resultList.appendChild(item);

      });


      // 모든 검색 결과가 보이도록 지도 범위를 조절합니다.

      if (places.length > 0) {

        map.setBounds(bounds);

      }

    }


    // 검색 버튼을 누르거나 Enter 키를 누르면 검색합니다.

    searchForm.addEventListener("submit", function (event) {

      event.preventDefault();

      searchPlaces(keywordInput.value);

    });


    // SDK가 준비된 뒤에 지도 초기화 함수를 실행합니다.

    if (window.kakao && window.kakao.maps) {

      kakao.maps.load(initializeMap);

    } else {

      showSdkError();

    }

  </script>
